# task 1
select order_id, sum(li.quantity * p.price) as total_price  from line_items li
join products p on li.product_id = p.product_id
group by order_id order by order_id limit 5;

# task 2
select customer_name, avg(t.total_price) average_order_price from customers c
join orders o on c.customer_id = o.customer_id
join
  (select li.order_id, sum(li.quantity * p.price) total_price from line_items li join products p on li.product_id = p.product_id group by li.order_id)
  as t
  on t.order_id = o.order_id
group by c.customer_id, c.customer_name
order by customer_name;
